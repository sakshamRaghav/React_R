
#include <webots/robot.h>
#include <webots/lidar.h>
#include <webots/motor.h>   

#include "robot_controller.h"   // also brings in the other headers
#include <cmath>
#include <iostream>
  
using namespace std;

// ---------------- settings: change these to match your robot ----------------
const char* LIDAR_NAME       = "lidar";
const char* LEFT_MOTOR_NAME  = "left wheel";    // Pioneer 3-DX names
const char* RIGHT_MOTOR_NAME = "right wheel";
const double WHEEL_RADIUS    = 0.0975;          // metres (Pioneer 3-DX)

const bool INDEX_GOES_LEFT = false;   // if LEFT and RIGHT are swapped, flip this

const int STUCK_TICKS = 1000000;


int main() {
    wb_robot_init();
    int time_step = (int)wb_robot_get_basic_time_step();

    // find the devices by their names
    WbDeviceTag lidar       = wb_robot_get_device(LIDAR_NAME);
    WbDeviceTag left_motor  = wb_robot_get_device(LEFT_MOTOR_NAME);
    WbDeviceTag right_motor = wb_robot_get_device(RIGHT_MOTOR_NAME);

    // 0 means Webots did not find a device with that name
    if (lidar == 0 || left_motor == 0 || right_motor == 0) {
        cerr << "Device not found. Check LIDAR_NAME and the motor names.\n";
        wb_robot_cleanup();
        return 1;
    }

    wb_lidar_enable(lidar, time_step);
    int n = wb_lidar_get_horizontal_resolution(lidar);

    // velocity control: no target position, we only set speeds
    wb_motor_set_position(left_motor,  INFINITY);
    wb_motor_set_position(right_motor, INFINITY);
    wb_motor_set_velocity(left_motor,  0.0);
    wb_motor_set_velocity(right_motor, 0.0);

    // the fastest the wheel is allowed to turn (rad/s)
    double max_wheel_speed = wb_motor_get_max_velocity(left_motor);

    // our own code
    SensorMonitor   sensors;
    RiskAssessment  risk;
    DecisionEngine  decision;
    RobotController robot;

    sensors.registerSensor(1, SensorDirection::FRONT, 0.05, 5.0);
    sensors.registerSensor(2, SensorDirection::REAR,  0.05, 5.0);
    sensors.registerSensor(3, SensorDirection::LEFT,  0.05, 5.0);
    sensors.registerSensor(4, SensorDirection::RIGHT, 0.05, 5.0);

    long long tick = 0;
    RobotAction last_action = RobotAction::STOP;

    // main loop: one pass = one simulation step
    while (wb_robot_step(time_step) != -1) {
        tick++;

        // 1. read the lidar and update our 4 sensors
        const float* scan = wb_lidar_get_range_image(lidar);
        bool moving = robot.getCurrentSpeed() > 0.0;
        sensors.updateFromLidar(scan, n, tick, n / 2, INDEX_GOES_LEFT,
                                moving, STUCK_TICKS);

        // 2. our code decides what to do
        RiskLevel   level  = risk.evaluate(sensors);
        RobotAction action = decision.decide(level);
        robot.execute(action);

        // 3. turn the speed (m/s) into wheel speed (rad/s) and set the motors
        double wheel_speed = robot.getCurrentSpeed() / WHEEL_RADIUS;
        if (wheel_speed > max_wheel_speed) {
            wheel_speed = max_wheel_speed;
        }
        wb_motor_set_velocity(left_motor,  wheel_speed);
        wb_motor_set_velocity(right_motor, wheel_speed);

        // print only when the action changes, so the console stays readable
        if (action != last_action) {
            cout << "tick " << tick
                 << " | risk "   << riskToString(level)
                 << " | action " << actionToString(action) << "\n";
            sensors.printStatus();
            last_action = action;
        }
    }

    wb_robot_cleanup();
    return 0;
}
